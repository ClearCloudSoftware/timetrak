> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Rebuild the tray popover as a "home base" with today's entries, one-or-two-click resume, header icons, description cap, and a daily total — plus fix dev-mode Dock icon and tray-anchored positioning.

**Architecture:** Frontend-only state-machine in `src/windows/tray/` (modes: `browse` / `new` / `edit`) backed by the existing frozen Rust commands. One new Tauri command (`open_window`) is added to let header icons open the Dashboard and Settings windows. macOS activation policy is set programmatically so dev mode also hides the Dock icon. `TrayIconEvent::Click` rect is plumbed into the popover positioning code.

**Tech Stack:** Tauri 2, React 18 + TypeScript, Tailwind, `@tanstack/react-query`, `lucide-react` (new dep), Vitest.

```rust
const POPOVER_W: f64 = 340.0;
const POPOVER_H: f64 = 440.0;

fn toggle_tray_window(app: &tauri::AppHandle, tray_rect: Option<tauri::Rect>) {
    let position = tray_rect.map(|r| compute_popover_position(r));

    if let Some(w) = app.get_webview_window(TRAY_WINDOW_LABEL) {
        if w.is_visible().unwrap_or(false) {
            let _ = w.hide();
            return;
        }
        if let Some(p) = position {
            let _ = w.set_position(p);
        }
        let _ = w.show();
        let _ = w.set_focus();
        return;
    }
    let mut builder = tauri::WebviewWindowBuilder::new(
        app,
        TRAY_WINDOW_LABEL,
        tauri::WebviewUrl::App("index.html?window=tray".into()),
    )
    .inner_size(POPOVER_W, POPOVER_H)
    .decorations(false)
    .resizable(false)
    .always_on_top(true)
    .skip_taskbar(true)
    .visible(true);
    if let Some(p) = position {
        builder = builder.position(p.x, p.y);
    }
    let _ = builder.build();
}

#[cfg(target_os = "macos")]
fn compute_popover_position(rect: tauri::Rect) -> tauri::PhysicalPosition<f64> {
    let x = rect.position.x as f64 + (rect.size.width as f64 / 2.0) - (POPOVER_W / 2.0);
    let y = rect.position.y as f64 + rect.size.height as f64 + 4.0;
    tauri::PhysicalPosition::new(x, y)
}

#[cfg(not(target_os = "macos"))]
fn compute_popover_position(rect: tauri::Rect) -> tauri::PhysicalPosition<f64> {
    // Windows: taskbar at the bottom by default, so position the popover above the tray icon.
    let x = rect.position.x as f64 + (rect.size.width as f64 / 2.0) - (POPOVER_W / 2.0);
    let y = rect.position.y as f64 - POPOVER_H - 4.0;
    tauri::PhysicalPosition::new(x, y)
}
```

> **Note:** The `inner_size(340.0, 440.0)` call already in the original `toggle_tray_window` is now keyed off the `POPOVER_W`/`POPOVER_H` constants. The position-passing call to `builder.position(p.x, p.y)` uses `f64`; verify Tauri 2's builder accepts `f64` (it does — `position` accepts `impl Into<LogicalPosition<f64>>`).

```rust
//! Window-management Tauri commands.

use tauri::{AppHandle, Manager, WebviewUrl, WebviewWindowBuilder};

use crate::error::{AppError, AppResult};

#[tauri::command]
pub fn open_window(app: AppHandle, name: String) -> AppResult<()> {
    if name != "dashboard" && name != "settings" {
        return Err(AppError::Invalid(format!("unknown window: {name}")));
    }
    if let Some(w) = app.get_webview_window(&name) {
        w.show().map_err(|e| AppError::Other(e.to_string()))?;
        w.set_focus().map_err(|e| AppError::Other(e.to_string()))?;
        return Ok(());
    }
    let title = match name.as_str() {
        "dashboard" => "TimeTrak — Dashboard",
        "settings" => "TimeTrak — Settings",
        _ => "TimeTrak",
    };
    WebviewWindowBuilder::new(
        &app,
        &name,
        WebviewUrl::App(format!("index.html?window={name}").into()),
    )
    .title(title)
    .inner_size(900.0, 600.0)
    .build()
    .map_err(|e| AppError::Other(e.to_string()))?;
    Ok(())
}
```

```ts
export function todayRangeUtc(): { startUtc: string; endUtc: string } {
  const now = new Date();
  const start = new Date(now);
  start.setHours(0, 0, 0, 0);
  const end = new Date(start);
  end.setDate(start.getDate() + 1);
  return { startUtc: start.toISOString(), endUtc: end.toISOString() };
}

export function formatDuration(totalSeconds: number): string {
  const safe = Math.max(0, Math.floor(totalSeconds));
  const h = Math.floor(safe / 3600);
  const m = Math.floor((safe % 3600) / 60);
  if (h > 0) return `${h}h ${m}m`;
  return `${m}m`;
}

export function formatStartTime(iso: string): string {
  const d = new Date(iso);
  const pad = (n: number) => String(n).padStart(2, '0');
  return `${pad(d.getHours())}:${pad(d.getMinutes())}`;
}
```

```tsx
import type { Category, Project, TimeEntry } from '../../types';
import { formatDuration, formatStartTime } from './format';

interface Props {
  entry: TimeEntry;
  categories: Category[];
  projects: Project[];
  onResume: (entry: TimeEntry) => void;
}

export function TodayEntryRow({ entry, categories, projects, onResume }: Props) {
  const cat = categories.find((c) => c.id === entry.category_id);
  const proj = entry.project_id ? projects.find((p) => p.id === entry.project_id) : null;
  const durationSeconds = entry.ended_at
    ? Math.max(0, (new Date(entry.ended_at).getTime() - new Date(entry.started_at).getTime()) / 1000)
    : 0;

  const handleActivate = () => onResume(entry);
  const handleKeyDown = (e: React.KeyboardEvent) => {
    if (e.key === 'Enter' || e.key === ' ') {
      e.preventDefault();
      handleActivate();
    }
  };

  return (
    <div
      role="button"
      tabIndex={0}
      onClick={handleActivate}
      onKeyDown={handleKeyDown}
      className="cursor-pointer rounded px-2 py-1.5 transition-colors hover:bg-gray-50 focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-blue-500"
    >
      <div className="flex items-center justify-between gap-2 text-sm">
        <div className="flex items-center gap-2 truncate">
          <span className="text-gray-500 tabular-nums">{formatStartTime(entry.started_at)}</span>
          <span
            aria-hidden
            className="inline-block h-2 w-2 shrink-0 rounded-full"
            style={{ background: cat?.color }}
          />
          <span className="truncate">{cat?.name ?? '?'}</span>
          {proj && <span className="truncate text-gray-500">· {proj.name}</span>}
        </div>
        <span className="shrink-0 text-gray-500 tabular-nums">{formatDuration(durationSeconds)}</span>
      </div>
      {entry.note && (
        <div className="truncate pl-12 text-xs text-gray-500">{entry.note}</div>
      )}
    </div>
  );
}
```

```tsx
import type { TimeEntry } from '../../types';
import { formatDuration } from './format';

interface Props {
  entries: TimeEntry[];
  runningEntry: TimeEntry | null;
  nowMs: number;
}

export function TodayFooter({ entries, runningEntry, nowMs }: Props) {
  let totalSeconds = 0;
  for (const e of entries) {
    if (e.id === runningEntry?.id) continue;
    if (!e.ended_at) continue;
    totalSeconds += (new Date(e.ended_at).getTime() - new Date(e.started_at).getTime()) / 1000;
  }
  if (runningEntry) {
    totalSeconds += Math.max(0, (nowMs - new Date(runningEntry.started_at).getTime()) / 1000);
  }
  return (
    <div className="border-t border-gray-200 px-3 py-2 text-sm font-medium">
      Today: {formatDuration(totalSeconds)}
    </div>
  );
}
```

```tsx
import { useEffect, useState } from 'react';
import type { Category, Project, TimeEntry } from '../../types';

interface Props {
  entry: TimeEntry;
  categories: Category[];
  projects: Project[];
  onStop: () => void;
  stopPending: boolean;
}

export function RunningBanner({ entry, categories, projects, onStop, stopPending }: Props) {
  const cat = categories.find((c) => c.id === entry.category_id);
  const proj = entry.project_id ? projects.find((p) => p.id === entry.project_id) : null;
  const [elapsed, setElapsed] = useState(() => elapsedSeconds(entry.started_at));
  useEffect(() => {
    const t = setInterval(() => setElapsed(elapsedSeconds(entry.started_at)), 1000);
    return () => clearInterval(t);
  }, [entry.started_at]);

  return (
    <div className="rounded-md border border-gray-200 bg-white p-3">
      <div className="text-xs font-medium uppercase tracking-wide text-gray-500">Running</div>
      <div className="mt-1 flex items-center gap-2 text-sm">
        <span
          aria-hidden
          className="inline-block h-2 w-2 shrink-0 rounded-full"
          style={{ background: cat?.color }}
        />
        <span className="truncate font-medium">{cat?.name ?? '?'}</span>
        {proj && <span className="truncate text-gray-500">· {proj.name}</span>}
      </div>
      {entry.note && (
        <div className="mt-0.5 truncate text-xs text-gray-500">{entry.note}</div>
      )}
      <div className="mt-2 flex items-center justify-between">
        <div
          aria-label={`Elapsed ${formatElapsed(elapsed)}`}
          className="font-mono text-2xl tabular-nums"
        >
          {formatElapsed(elapsed)}
        </div>
        <button
          type="button"
          onClick={onStop}
          disabled={stopPending}
          className="rounded border border-gray-200 px-3 py-1 text-sm text-red-600 transition-colors hover:bg-red-50 focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-red-500 disabled:cursor-not-allowed disabled:text-gray-400"
        >
          Stop
        </button>
      </div>
    </div>
  );
}

function elapsedSeconds(startedAtIso: string): number {
  return Math.max(0, Math.floor((Date.now() - new Date(startedAtIso).getTime()) / 1000));
}

function formatElapsed(s: number): string {
  const h = Math.floor(s / 3600);
  const m = Math.floor((s % 3600) / 60);
  const sec = s % 60;
  return `${String(h).padStart(2, '0')}:${String(m).padStart(2, '0')}:${String(sec).padStart(2, '0')}`;
}
```

```tsx
import { useEffect, useMemo, useState } from 'react';
import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query';
import * as api from '../../lib/api';
import { onTimerChanged, onEntriesChanged } from '../../lib/events';
import { qk } from '../../lib/query';
import type { Id, TimeEntry } from '../../types';
import { HeaderIcons } from './HeaderIcons';
import { RunningBanner } from './RunningBanner';
import { TodayList } from './TodayList';
import { TodayFooter } from './TodayFooter';
import { InlineTimerForm } from './InlineTimerForm';
import { TodayEntryRow } from './TodayEntryRow';
import { todayRangeUtc } from './format';

type Mode =
  | { kind: 'browse' }
  | { kind: 'new' }
  | { kind: 'edit'; sourceEntry: TimeEntry };

export function TrayPopover() {
  const qc = useQueryClient();
  const range = useMemo(() => todayRangeUtc(), []);

  const categories = useQuery({ queryKey: qk.categories, queryFn: api.listCategories });
  const projects = useQuery({ queryKey: qk.projects, queryFn: api.listProjects });
  const timer = useQuery({ queryKey: qk.timerState, queryFn: api.getTimerState });
  const entries = useQuery({
    queryKey: qk.entries(range.startUtc, range.endUtc),
    queryFn: () => api.listEntries(range.startUtc, range.endUtc),
  });

  useEffect(() => {
    const unsubs: Array<() => void> = [];
    onTimerChanged(() => {
      qc.invalidateQueries({ queryKey: qk.timerState });
      qc.invalidateQueries({ queryKey: ['entries'] });
    }).then((u) => unsubs.push(u));
    onEntriesChanged(() => qc.invalidateQueries({ queryKey: ['entries'] })).then((u) => unsubs.push(u));
    return () => { unsubs.forEach((u) => u()); };
  }, [qc]);

  // Live nowMs tick (1s) for the footer total
  const [nowMs, setNowMs] = useState(Date.now());
  useEffect(() => {
    const t = setInterval(() => setNowMs(Date.now()), 1000);
    return () => clearInterval(t);
  }, []);

  const [mode, setMode] = useState<Mode>({ kind: 'browse' });
  const [errorMessage, setErrorMessage] = useState<string | null>(null);

  const running = timer.data?.running ?? null;
  const cats = categories.data ?? [];
  const projs = projects.data ?? [];
  const allEntries = entries.data ?? [];

  const startOrSwitch = useMutation({
    mutationFn: async (v: { categoryId: Id; projectId: Id | null; description: string | null }) => {
      if (running) {
        return api.switchTimer(v.categoryId, v.projectId, v.description);
      }
      return api.startTimer(v.categoryId, v.projectId, v.description);
    },
    onSuccess: () => {
      qc.invalidateQueries({ queryKey: qk.timerState });
      qc.invalidateQueries({ queryKey: ['entries'] });
      setMode({ kind: 'browse' });
      setErrorMessage(null);
    },
    onError: (e) => setErrorMessage(String(e)),
  });

  const stop = useMutation({
    mutationFn: () => api.stopTimer(),
    onSuccess: () => {
      qc.invalidateQueries({ queryKey: qk.timerState });
      qc.invalidateQueries({ queryKey: ['entries'] });
      setErrorMessage(null);
    },
    onError: (e) => setErrorMessage(String(e)),
  });

  const handleResume = (entry: TimeEntry) => setMode({ kind: 'edit', sourceEntry: entry });
  const handleNew = () => setMode({ kind: 'new' });
  const handleCancel = () => setMode({ kind: 'browse' });
  const handleSubmit = (v: { categoryId: Id; projectId: Id | null; description: string | null }) =>
    startOrSwitch.mutate(v);

  return (
    <div className="flex h-full flex-col bg-white text-sm">
      <header className="flex items-center justify-between border-b border-gray-200 px-3 py-2">
        <div className="text-xs font-semibold uppercase tracking-widest text-gray-400">TimeTrak</div>
        <HeaderIcons onError={setErrorMessage} />
      </header>

      <div className="flex-1 overflow-auto px-3 py-2 space-y-3">
        {running && (
          <RunningBanner
            entry={running}
            categories={cats}
            projects={projs}
            onStop={() => stop.mutate()}
            stopPending={stop.isPending}
          />
        )}

        {mode.kind === 'new' ? (
          <InlineTimerForm
            initial={{ categoryId: null, projectId: null, description: '' }}
            categories={cats}
            projects={projs}
            onCancel={handleCancel}
            onSubmit={handleSubmit}
            submitting={startOrSwitch.isPending}
          />
        ) : (
          <button
            type="button"
            onClick={handleNew}
            className="w-full rounded border border-dashed border-gray-300 px-3 py-2 text-sm text-gray-600 transition-colors hover:border-gray-400 hover:text-gray-900 focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-blue-500"
          >
            ＋ New timer
          </button>
        )}

        <div>
          <div className="mb-1 px-2 text-xs font-medium uppercase tracking-wide text-gray-500">
            Today's timers
          </div>
          {mode.kind === 'edit' ? (
            <div className="space-y-2">
              <TodayEntryRow
                entry={mode.sourceEntry}
                categories={cats}
                projects={projs}
                onResume={() => {}}
              />
              <InlineTimerForm
                initial={{
                  categoryId: mode.sourceEntry.category_id,
                  projectId: mode.sourceEntry.project_id,
                  description: mode.sourceEntry.note ?? '',
                }}
                categories={cats}
                projects={projs}
                onCancel={handleCancel}
                onSubmit={handleSubmit}
                submitting={startOrSwitch.isPending}
              />
              <TodayList
                entries={allEntries.filter((e) => e.id !== mode.sourceEntry.id)}
                categories={cats}
                projects={projs}
                runningId={running?.id ?? null}
                onResume={handleResume}
              />
            </div>
          ) : (
            <TodayList
              entries={allEntries}
              categories={cats}
              projects={projs}
              runningId={running?.id ?? null}
              onResume={handleResume}
            />
          )}
        </div>

        {errorMessage && (
          <div className="rounded-md border border-red-200 bg-red-50 px-3 py-2 text-xs text-red-700">
            {errorMessage}
          </div>
        )}
      </div>

      <TodayFooter entries={allEntries} runningEntry={running} nowMs={nowMs} />
    </div>
  );
}
```
